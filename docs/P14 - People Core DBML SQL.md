-- ========================================================
-- People & Identity (People Core) Domain
-- Assumption: clean schema (tables do NOT yet exist).
-- ========================================================

BEGIN;

-- --------------------------------------------------------
-- 1. System / Shared stubs (for FK references)
-- --------------------------------------------------------

CREATE TABLE sys_tenant (
    ten_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ten_name VARCHAR(200) NOT NULL
);

COMMENT ON TABLE sys_tenant IS
    'Tenant table (stub; full definition in System/Shared domain).';


-- --------------------------------------------------------
-- 2. Global person identity (no tenant)
-- --------------------------------------------------------

CREATE TABLE pid_person (
    per_id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    per_first_name       VARCHAR(100) NOT NULL,
    per_middle_name      VARCHAR(100),
    per_last_name        VARCHAR(150) NOT NULL,
    per_display_name     VARCHAR(255),
    per_gender_code      VARCHAR(50),
    per_date_of_birth    DATE,
    per_nationality_code VARCHAR(10),

    per_created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
    per_created_by       BIGINT,
    per_updated_at       TIMESTAMPTZ,
    per_updated_by       BIGINT,
    per_deleted_at       TIMESTAMPTZ,
    per_deleted_by       BIGINT
);

COMMENT ON TABLE pid_person IS
    'Global person identity shared across tenants/platforms.';
COMMENT ON COLUMN pid_person.per_display_name IS
    'Denormalised display/full name for search & UI.';


-- --------------------------------------------------------
-- 3. Cross-domain stub – Org Position Assignment
--    (Full definition lives in Org & Jobs / Work Lattice domain.)
// --------------------------------------------------------

CREATE TABLE org_position_assignment (
    pas_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY
);

COMMENT ON TABLE org_position_assignment IS
    'Stub for cross-domain FK from pid_emp_contract. Full definition in Org & Jobs domain.';


-- --------------------------------------------------------
-- 4. Person contacts (email, phone, address etc.) – global
-- --------------------------------------------------------

CREATE TABLE pid_person_contact (
    pco_id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    pco_per_id            BIGINT NOT NULL,
    pco_contact_type_code VARCHAR(50)  NOT NULL,  -- EMAIL, MOBILE, PHONE, ADDRESS, OTHER
    pco_contact_value     VARCHAR(255) NOT NULL,
    pco_is_primary        BOOLEAN      NOT NULL DEFAULT FALSE,
    pco_label             VARCHAR(50),
    pco_country_code      VARCHAR(10),

    pco_created_at        TIMESTAMPTZ  NOT NULL DEFAULT now(),
    pco_created_by        BIGINT,
    pco_updated_at        TIMESTAMPTZ,
    pco_updated_by        BIGINT,
    pco_deleted_at        TIMESTAMPTZ,
    pco_deleted_by        BIGINT,

    CONSTRAINT fk_pco_per
        FOREIGN KEY (pco_per_id) REFERENCES pid_person (per_id)
);

COMMENT ON TABLE pid_person_contact IS
    'Multiple contact methods per person; primary flag per type.';
COMMENT ON COLUMN pid_person_contact.pco_is_primary IS
    'Indicates primary contact of this type for the person.';


-- --------------------------------------------------------
-- 5. Official identifiers – global
-- --------------------------------------------------------

CREATE TABLE pid_person_identifier (
    idn_id                   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    idn_per_id               BIGINT       NOT NULL,
    idn_identifier_type_code VARCHAR(50)  NOT NULL, -- NATIONAL_ID, PASSPORT, TAX_PIN, etc.
    idn_identifier_value     VARCHAR(100) NOT NULL,
    idn_country_code         VARCHAR(10),
    idn_valid_from           DATE,
    idn_valid_to             DATE,

    idn_created_at           TIMESTAMPTZ  NOT NULL DEFAULT now(),
    idn_created_by           BIGINT,
    idn_updated_at           TIMESTAMPTZ,
    idn_updated_by           BIGINT,
    idn_deleted_at           TIMESTAMPTZ,
    idn_deleted_by           BIGINT,

    CONSTRAINT fk_idn_per
        FOREIGN KEY (idn_per_id) REFERENCES pid_person (per_id),

    CONSTRAINT uq_idn_type_country_value
        UNIQUE (idn_identifier_type_code, idn_country_code, idn_identifier_value)
);

COMMENT ON TABLE pid_person_identifier IS
    'Multiple identifiers per person; type+country+value pair is unique.';


-- --------------------------------------------------------
-- 6. Employee per tenant – tenant-scoped
-- --------------------------------------------------------

CREATE TABLE pid_employee (
    emp_id                          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    emp_tenant_id                   BIGINT      NOT NULL,
    emp_per_id                      BIGINT      NOT NULL,
    emp_employee_number             VARCHAR(50) NOT NULL, -- tenant-specific employee code
    emp_hire_date                   DATE,
    emp_employment_type_code        VARCHAR(50),          -- PERMANENT, FIXED_TERM, CASUAL, INTERN, CONSULTANT
    emp_current_status_code         VARCHAR(50),          -- PLANNED, ACTIVE, PROBATION, SUSPENDED, EXITED
    emp_current_status_effective_date DATE,

    emp_created_at                  TIMESTAMPTZ NOT NULL DEFAULT now(),
    emp_created_by                  BIGINT,
    emp_updated_at                  TIMESTAMPTZ,
    emp_updated_by                  BIGINT,
    emp_deleted_at                  TIMESTAMPTZ,
    emp_deleted_by                  BIGINT,

    CONSTRAINT fk_emp_tenant
        FOREIGN KEY (emp_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_emp_person
        FOREIGN KEY (emp_per_id)    REFERENCES pid_person (per_id),

    CONSTRAINT uq_emp_tenant_number
        UNIQUE (emp_tenant_id, emp_employee_number),
    CONSTRAINT uq_emp_tenant_person
        UNIQUE (emp_tenant_id, emp_per_id)
);

COMMENT ON TABLE pid_employee IS
    'Employee record per tenant, linked to global person; one employee row per (tenant, person).';


-- --------------------------------------------------------
-- 7. Employment contract – per employee per tenant
-- --------------------------------------------------------

CREATE TABLE pid_emp_contract (
    ctr_id                      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ctr_tenant_id               BIGINT       NOT NULL,
    ctr_emp_id                  BIGINT       NOT NULL,
    ctr_contract_type_code      VARCHAR(50)  NOT NULL, -- PERMANENT, FIXED_TERM, INTERNSHIP etc.
    ctr_start_date              DATE         NOT NULL,
    ctr_end_date                DATE,
    ctr_probation_end_date      DATE,
    ctr_standard_hours_per_week DECIMAL(5,2),
    ctr_standard_days_per_week  DECIMAL(3,2),
    ctr_status_code             VARCHAR(50)  NOT NULL, -- DRAFT, ACTIVE, ENDED etc.
    ctr_prm_pas_id              BIGINT,                -- primary position assignment (cross-domain)

    ctr_created_at              TIMESTAMPTZ  NOT NULL DEFAULT now(),
    ctr_created_by              BIGINT,
    ctr_updated_at              TIMESTAMPTZ,
    ctr_updated_by              BIGINT,
    ctr_deleted_at              TIMESTAMPTZ,
    ctr_deleted_by              BIGINT,

    CONSTRAINT fk_ctr_tenant
        FOREIGN KEY (ctr_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_ctr_emp
        FOREIGN KEY (ctr_emp_id)    REFERENCES pid_employee (emp_id),
    CONSTRAINT fk_ctr_prm_pas
        FOREIGN KEY (ctr_prm_pas_id) REFERENCES org_position_assignment (pas_id)
);

COMMENT ON TABLE pid_emp_contract IS
    'Supports multiple contracts over time per employee; links to primary position assignment.';
COMMENT ON COLUMN pid_emp_contract.ctr_prm_pas_id IS
    'Primary position assignment ID from Org & Jobs domain.';


-- --------------------------------------------------------
-- 8. Employment status history – per employee per tenant
-- --------------------------------------------------------

CREATE TABLE pid_emp_status_history (
    esh_id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    esh_tenant_id       BIGINT      NOT NULL,
    esh_emp_id          BIGINT      NOT NULL,
    esh_status_code     VARCHAR(50) NOT NULL, -- ACTIVE, PROBATION, SUSPENDED, EXITED, etc.
    esh_effective_date  DATE        NOT NULL,
    esh_reason_code     VARCHAR(50),
    esh_reason_note     TEXT,

    esh_created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    esh_created_by      BIGINT,
    esh_updated_at      TIMESTAMPTZ,
    esh_updated_by      BIGINT,
    esh_deleted_at      TIMESTAMPTZ,
    esh_deleted_by      BIGINT,

    CONSTRAINT fk_esh_tenant
        FOREIGN KEY (esh_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_esh_emp
        FOREIGN KEY (esh_emp_id)    REFERENCES pid_employee (emp_id)
);

COMMENT ON TABLE pid_emp_status_history IS
    'Chronological employment status timeline per employee.';


-- --------------------------------------------------------
-- 9. Dependents – per employee per tenant
-- --------------------------------------------------------

CREATE TABLE pid_dependent (
    dep_id                       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    dep_tenant_id                BIGINT       NOT NULL,
    dep_emp_id                   BIGINT       NOT NULL,
    dep_name                     VARCHAR(200),
    dep_relationship_code        VARCHAR(50)  NOT NULL, -- SPOUSE, CHILD, PARENT, OTHER
    dep_date_of_birth            DATE,
    dep_included_in_health_cover BOOLEAN      NOT NULL DEFAULT FALSE,
    dep_wellness_eligible        BOOLEAN      NOT NULL DEFAULT FALSE,

    dep_created_at               TIMESTAMPTZ  NOT NULL DEFAULT now(),
    dep_created_by               BIGINT,
    dep_updated_at               TIMESTAMPTZ,
    dep_updated_by               BIGINT,
    dep_deleted_at               TIMESTAMPTZ,
    dep_deleted_by               BIGINT,

    CONSTRAINT fk_dep_tenant
        FOREIGN KEY (dep_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_dep_emp
        FOREIGN KEY (dep_emp_id)    REFERENCES pid_employee (emp_id)
);

COMMENT ON TABLE pid_dependent IS
    'Dependents used for wellness and high-level eligibility, not full benefits admin.';


-- --------------------------------------------------------
-- 10. Qualifications – per person (global)
// --------------------------------------------------------

CREATE TABLE pid_qualification (
    qlf_id                      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    qlf_per_id                  BIGINT       NOT NULL,
    qlf_qualification_type_code VARCHAR(50)  NOT NULL, -- EDUCATION, PROFESSIONAL etc.
    qlf_institution             VARCHAR(255),
    qlf_qualification_name      VARCHAR(255) NOT NULL,
    qlf_level_code              VARCHAR(50),           -- DIPLOMA, DEGREE, MASTERS etc.
    qlf_completion_year         INT,

    qlf_created_at              TIMESTAMPTZ  NOT NULL DEFAULT now(),
    qlf_created_by              BIGINT,
    qlf_updated_at              TIMESTAMPTZ,
    qlf_updated_by              BIGINT,
    qlf_deleted_at              TIMESTAMPTZ,
    qlf_deleted_by              BIGINT,

    CONSTRAINT fk_qlf_per
        FOREIGN KEY (qlf_per_id) REFERENCES pid_person (per_id)
);

COMMENT ON TABLE pid_qualification IS
    'High-level education & professional qualifications for context in performance & development.';


-- --------------------------------------------------------
-- 11. Prior employment history – per person (global)
-- --------------------------------------------------------

CREATE TABLE pid_employment_history (
    peh_id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    peh_per_id        BIGINT       NOT NULL,
    peh_employer_name VARCHAR(255) NOT NULL,
    peh_role_title    VARCHAR(255),
    peh_start_date    DATE,
    peh_end_date      DATE,
    peh_summary       TEXT,

    peh_created_at    TIMESTAMPTZ  NOT NULL DEFAULT now(),
    peh_created_by    BIGINT,
    peh_updated_at    TIMESTAMPTZ,
    peh_updated_by    BIGINT,
    peh_deleted_at    TIMESTAMPTZ,
    peh_deleted_by    BIGINT,

    CONSTRAINT fk_peh_per
        FOREIGN KEY (peh_per_id) REFERENCES pid_person (per_id)
);

COMMENT ON TABLE pid_employment_history IS
    'Prior employment summaries for contextual use in PeopleWell and performance reviews.';


-- --------------------------------------------------------
-- 12. Wellness profile – per employee per tenant
-- --------------------------------------------------------

CREATE TABLE pid_wellness_profile (
    wep_id                     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    wep_tenant_id              BIGINT      NOT NULL,
    wep_emp_id                 BIGINT      NOT NULL,
    wep_consent_flag           BOOLEAN     NOT NULL DEFAULT FALSE,
    wep_preferred_channel_code VARCHAR(50),            -- EMAIL, SMS, WHATSAPP, APP etc.
    wep_last_zhep_sync_at      TIMESTAMPTZ,

    wep_created_at             TIMESTAMPTZ NOT NULL DEFAULT now(),
    wep_created_by             BIGINT,
    wep_updated_at             TIMESTAMPTZ,
    wep_updated_by             BIGINT,
    wep_deleted_at             TIMESTAMPTZ,
    wep_deleted_by             BIGINT,

    CONSTRAINT fk_wep_tenant
        FOREIGN KEY (wep_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_wep_emp
        FOREIGN KEY (wep_emp_id)    REFERENCES pid_employee (emp_id),

    CONSTRAINT uq_wep_emp_id
        UNIQUE (wep_emp_id)
);

COMMENT ON TABLE pid_wellness_profile IS
    'Per-employee wellness engagement profile (consent, channels, sync marker).';
COMMENT ON COLUMN pid_wellness_profile.wep_consent_flag IS
    'Indicates whether employee has given consent for wellness engagement.';


-- --------------------------------------------------------
-- 13. Wellness profile tags – per wellness profile per tenant
-- --------------------------------------------------------

CREATE TABLE pid_wellness_profile_tag (
    wpt_id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    wpt_tenant_id       BIGINT       NOT NULL,
    wpt_wep_id          BIGINT       NOT NULL,
    wpt_tag_code        VARCHAR(100) NOT NULL, -- arbitrary, opaque engagement tags
    wpt_source_system   VARCHAR(50)  NOT NULL, -- e.g. ZHEP, MANUAL
    wpt_first_seen_at   TIMESTAMPTZ  NOT NULL,
    wpt_last_updated_at TIMESTAMPTZ  NOT NULL,
    wpt_is_active       BOOLEAN      NOT NULL DEFAULT TRUE,

    wpt_created_at      TIMESTAMPTZ  NOT NULL DEFAULT now(),
    wpt_created_by      BIGINT,
    wpt_updated_at      TIMESTAMPTZ,
    wpt_updated_by      BIGINT,
    wpt_deleted_at      TIMESTAMPTZ,
    wpt_deleted_by      BIGINT,

    CONSTRAINT fk_wpt_tenant
        FOREIGN KEY (wpt_tenant_id) REFERENCES sys_tenant (ten_id),
    CONSTRAINT fk_wpt_wep
        FOREIGN KEY (wpt_wep_id)    REFERENCES pid_wellness_profile (wep_id),

    CONSTRAINT uq_wpt_profile_tag
        UNIQUE (wpt_tenant_id, wpt_wep_id, wpt_tag_code)
);

COMMENT ON TABLE pid_wellness_profile_tag IS
    'Engagement/wellness tags attached to a wellness profile, with history and source.';


-- --------------------------------------------------------
-- Indexes (non-unique)
// --------------------------------------------------------

-- pid_person
CREATE INDEX idx_per_last_first
    ON pid_person (per_last_name, per_first_name);

CREATE INDEX idx_per_display_name
    ON pid_person (per_display_name);

-- pid_person_contact
CREATE INDEX idx_pco_per_id
    ON pid_person_contact (pco_per_id);

CREATE INDEX idx_pco_person_type_primary
    ON pid_person_contact (pco_per_id, pco_contact_type_code, pco_is_primary);

-- pid_person_identifier
CREATE INDEX idx_idn_per_id
    ON pid_person_identifier (idn_per_id);

-- pid_employee
CREATE INDEX idx_emp_tenant_id
    ON pid_employee (emp_tenant_id);

-- pid_emp_contract
CREATE INDEX idx_ctr_tenant_id
    ON pid_emp_contract (ctr_tenant_id);

CREATE INDEX idx_ctr_emp_id
    ON pid_emp_contract (ctr_emp_id);

CREATE INDEX idx_ctr_prm_pas_id
    ON pid_emp_contract (ctr_prm_pas_id);

CREATE INDEX idx_ctr_tenant_emp_status
    ON pid_emp_contract (ctr_tenant_id, ctr_emp_id, ctr_status_code);

-- pid_emp_status_history
CREATE INDEX idx_esh_tenant_id
    ON pid_emp_status_history (esh_tenant_id);

CREATE INDEX idx_esh_tenant_emp_date
    ON pid_emp_status_history (esh_tenant_id, esh_emp_id, esh_effective_date);

-- pid_dependent
CREATE INDEX idx_dep_tenant_emp
    ON pid_dependent (dep_tenant_id, dep_emp_id);

-- pid_qualification
CREATE INDEX idx_qlf_per_id
    ON pid_qualification (qlf_per_id);

-- pid_employment_history
CREATE INDEX idx_peh_per_id
    ON pid_employment_history (peh_per_id);

-- pid_wellness_profile
CREATE INDEX idx_wep_tenant_id
    ON pid_wellness_profile (wep_tenant_id);

-- pid_wellness_profile_tag
CREATE INDEX idx_wpt_tenant_id
    ON pid_wellness_profile_tag (wpt_tenant_id);

CREATE INDEX idx_wpt_wep_id
    ON pid_wellness_profile_tag (wpt_wep_id);

COMMIT;
